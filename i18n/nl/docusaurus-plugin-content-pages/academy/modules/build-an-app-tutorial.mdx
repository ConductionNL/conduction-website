import {Prerequisites, PrerequisiteItem, Outcomes, Outcome, ModulePage} from '@conduction/docusaurus-preset/components';

<ModulePage module="build-an-app-tutorial">

Deze module bouwt PetStore, het canonieke OpenAPI pet-store voorbeeld, op de volledige Conduction-stack. Je scaffoldt de app, declareert de schema's, koppelt het met één config-blok aan de NC Calendar, laagt xWiki-verzorgingsgidsen erover, en publiceert naar de Conduction app-store. Elk onderdeel is zelfstandig. Sla delen over die je al kent.

<Outcomes title="Wat je bouwt">
  <Outcome>Een werkende PetStore Nextcloud-app in je lokale omgeving.</Outcome>
  <Outcome>Drie OpenRegister-schema's (pet, order, category) gedeclareerd in één manifest.</Outcome>
  <Outcome>NC Calendar-integratie zonder een controller of event listener te schrijven.</Outcome>
  <Outcome>Per-categorie verzorgingsgidsen uit xWiki, verpakt en gepubliceerd.</Outcome>
</Outcomes>

<Prerequisites title="Voor je begint">
  <PrerequisiteItem>Een Nextcloud op localhost (zie de [Run Nextcloud locally](/academy/run-nextcloud-locally) tutorial).</PrerequisiteItem>
  <PrerequisiteItem>OpenRegister geïnstalleerd en draaiend.</PrerequisiteItem>
  <PrerequisiteItem>Node 20 en PHP 8.3 lokaal, de standaard Conduction dev-stack.</PrerequisiteItem>
</Prerequisites>

</ModulePage>
