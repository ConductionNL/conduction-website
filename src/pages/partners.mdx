import {Section, SectionHead, PartnerDirectory, getApp} from '@conduction/docusaurus-preset/components';
import {usePartners, useBecomePartner} from '@site/src/data/partners-catalog';
import PageHeading from '@site/src/components/PageHeading';
import {useLocation} from '@docusaurus/router';
import Link from '@docusaurus/Link';

export function PartnerDirectoryBlock() {
  const partners = usePartners();
  const becomePartner = useBecomePartner();
  const {search} = useLocation();
  // Deeplink: /partners/?app=pipelinq or /partners/?solution=woo preselects
  // the offering. Product pages' "Request a demo from a partner" CTAs carry
  // this parameter, so the visitor lands on the partners that actually ship
  // the app they came from.
  const params = new URLSearchParams(search);
  const appSlug = params.get('app');
  const solutionSlug = params.get('solution');
  const appName = appSlug ? (getApp(appSlug)?.name || appSlug) : null;
  const shown = appName
    ? partners.filter(p => (p.apps || []).some(a => a.toLowerCase() === appName.toLowerCase()))
    : solutionSlug
      ? partners.filter(p => (p.solutions || []).some(s => s.toLowerCase() === solutionSlug.toLowerCase()))
      : partners;
  // Fall back to the full directory when the filter matches nobody, so a
  // stale or unknown slug never produces an empty page.
  const effective = shown.length > 0 ? shown : partners;
  const filterLabel = shown.length > 0 ? (appName || solutionSlug) : null;
  return (
    <>
      {filterLabel ? (
        <p style={{margin: '0 0 var(--space-4)'}}>
          Showing partners for <strong>{filterLabel}</strong>. <Link to="/partners">Show all partners</Link>
        </p>
      ) : null}
      <PartnerDirectory
        partners={effective}
        becomePartner={becomePartner}
        gridColumns={3}
      />
    </>
  );
}

<PageHeading>Implementation, hosting, integration.</PageHeading>

<Section spacing="default">

    eyebrow="Partners"

When you need help with hosting, custom integrations, or a regulated rollout, our certified partners take it from there. They run the same <span className="next-blue">Nextcloud</span> as you and ship on the same apps.</>}
  />

  <PartnerDirectoryBlock />
</Section>
